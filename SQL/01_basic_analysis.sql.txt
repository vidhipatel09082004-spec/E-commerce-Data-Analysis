Q17 What percentage of customers are repeat customers?

SELECT
    ROUND(
        (
            COUNT(*) * 100.0 /
            (SELECT COUNT(*) FROM customers)
        ),
        2
    ) AS retention_rate
FROM (
    SELECT
        customer_id
    FROM orders
    GROUP BY customer_id
    HAVING COUNT(*) > 1
) AS repeat_customers;

+----------------+
| retention_rate |
+----------------+
|          96.10 |
+----------------+

Q22 Which single day generated the highest revenue?

SELECT
    orders.order_date,
    SUM(orders.quantity * products.price) AS daily_revenue
FROM orders
JOIN products
    ON orders.product_id = products.product_id
GROUP BY orders.order_date
ORDER BY daily_revenue DESC
LIMIT 1;
